CREATE OR REPLACE PROCEDURE checknumber(n IN NUMBER) IS
BEGIN
  IF MOD(n, 2) = 0 THEN
    DBMS_OUTPUT.PUT_LINE(n || ' is Even');
  ELSE
    DBMS_OUTPUT.PUT_LINE(n || ' is Odd');
  END IF;
END;
/



CREATE OR REPLACE PROCEDURE find_largest(a IN NUMBER, b IN NUMBER, c IN NUMBER) IS
BEGIN
  IF a >= b AND a >= c THEN
    DBMS_OUTPUT.PUT_LINE('Largest is: ' || a);
  ELSIF b >= a AND b >= c THEN
    DBMS_OUTPUT.PUT_LINE('Largest is: ' || b);
  ELSE
    DBMS_OUTPUT.PUT_LINE('Largest is: ' || c);
  END IF;
END;
/




CREATE OR REPLACE PROCEDURE sum_and_average IS
  total   NUMBER := 0;
  average NUMBER;
BEGIN
  FOR i IN 1..10 LOOP
    total := total + i;
  END LOOP;
  average := total / 10;
  DBMS_OUTPUT.PUT_LINE('Sum: ' || total);
  DBMS_OUTPUT.PUT_LINE('Average: ' || average);
END;
/



CREATE OR REPLACE PROCEDURE calculator(a IN NUMBER, b IN NUMBER, op IN CHAR) IS
  result NUMBER;
BEGIN
  CASE op
    WHEN '+' THEN result := a + b;
    WHEN '-' THEN result := a - b;
    WHEN '*' THEN result := a * b;
    WHEN '/' THEN
      IF b != 0 THEN
        result := a / b;
      ELSE
        DBMS_OUTPUT.PUT_LINE('Error: Division by zero!');
        RETURN;
      END IF;
    ELSE
      DBMS_OUTPUT.PUT_LINE('Invalid operator!');
      RETURN;
  END CASE;
  DBMS_OUTPUT.PUT_LINE('Result: ' || result);
END;
/



CREATE OR REPLACE PROCEDURE create_student_table IS
BEGIN
  EXECUTE IMMEDIATE '
    CREATE TABLE student (
      id   NUMBER PRIMARY KEY,
      name VARCHAR2(50),
      max  NUMBER
    )';
  DBMS_OUTPUT.PUT_LINE('Table student created successfully.');
EXCEPTION
  WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
END;
/

CREATE OR REPLACE PROCEDURE multiplication_table(n IN NUMBER) IS
BEGIN
  FOR i IN 1..10 LOOP
    DBMS_OUTPUT.PUT_LINE(n || ' x ' || i || ' = ' || (n * i));
  END LOOP;
END;
/





CREATE OR REPLACE PROCEDURE display_odd_numbers IS
BEGIN
  FOR i IN 1..20 LOOP
    IF MOD(i, 2) != 0 THEN
      DBMS_OUTPUT.PUT_LINE(i);
    END IF;
  END LOOP;
END;
/
